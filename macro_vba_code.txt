Attribute VB_Name = "Module17"
Sub CalculMilieux_Moyenne_Seule()

    ' Exclure la feuille PRE construction
    If ActiveSheet.Name = "PRE construction" Then
        MsgBox "Cette macro ne s'applique pas à la feuille PRE construction.", vbExclamation
        Exit Sub
    End If

    Dim ws As Worksheet
    Dim i As Long, lastRow As Long
    Dim Cm As Variant, Cr As Variant, Cf As Double
    Dim Vm As Double, Vr As Double
    
    Vm = 1000
    Vr = 100
    
    Set ws = ActiveSheet
    lastRow = ws.Cells(ws.Rows.Count, "A").End(xlUp).Row
    
    For i = 2 To lastRow
        Cm = ws.Cells(i, 6).Value
        Cr = ws.Cells(i, 10).Value
        
        If Not IsNumeric(Cm) Then Cm = 0
        If Not IsNumeric(Cr) Then Cr = 0
        
        Cf = (Cm * Vm + Cr * Vr) / (Vm + Vr)
        
        ws.Cells(i, 11).Value = Round(Cf, 3)
    Next i
    
    MsgBox "Calcul des moyennes pondérées terminé, résultats en colonne K.", vbInformation

End Sub
